
#version 330 core

#define MAX_DIRECTIONAL_LIGHTS 4
#define MAX_POINT_LIGHTS 16
#define MAX_SPOT_LIGHTS 8

struct DirectionalLight
{
    vec3 direction;
    vec3 ambient;
    vec3 diffuse;
    vec3 specular;
    float intensity;
};

struct PointLight
{
    vec3 position;
    vec3 ambient;
    vec3 diffuse;
    vec3 specular;

    float constant;
    float linear;
    float quadratic;
    float intensity;
};

struct SpotLight
{
    vec3 position;
    vec3 direction;

    float innerAngle;
    float outerAngle;

    vec3 ambient;
    vec3 diffuse;
    vec3 specular;

    float constant;
    float linear;
    float quadratic;
    float intensity;
};

uniform DirectionalLight directionalLights[MAX_DIRECTIONAL_LIGHTS];
uniform PointLight pointLights[MAX_POINT_LIGHTS];
uniform SpotLight spotLights[MAX_SPOT_LIGHTS];

uniform int numDirectionalLights;
uniform int numPointLights;
uniform int numSpotLights;

uniform vec3 viewPos;
uniform vec3 materialColor;

uniform sampler2D texture_diffuse1;
uniform bool useTexture;

in VS_OUT
{
    vec3 FragPos;
    vec3 Normal;
    vec2 TexCoords;
} fs_in;

out vec4 FragColor;


// Directional light
vec3 CalculateDirectionalLight(
    DirectionalLight light,
    vec3 normal,
    vec3 viewDir,
    vec3 albedo)
{
    vec3 lightDir = normalize(-light.direction);

    float diff = max(dot(normal, lightDir), 0.0);

    vec3 reflectDir = reflect(-lightDir, normal);
    float spec = pow(
        max(dot(viewDir, reflectDir), 0.0),
        32.0
    );

    vec3 ambient = light.ambient * albedo;
    vec3 diffuse = light.diffuse * diff * albedo;
    vec3 specular = light.specular * spec;

    return (ambient + diffuse + specular) * light.intensity;
}


// Point light
vec3 CalculatePointLight(
    PointLight light,
    vec3 normal,
    vec3 viewDir,
    vec3 albedo)
{
    vec3 lightOffset = light.position - fs_in.FragPos;
    float distanceToLight = length(lightOffset);
    vec3 lightDir = normalize(lightOffset);

    float diff = max(dot(normal, lightDir), 0.0);

    vec3 reflectDir = reflect(-lightDir, normal);
    float spec = pow(
        max(dot(viewDir, reflectDir), 0.0),
        32.0
    );

    float attenuation = 1.0 /
        (light.constant +
         light.linear * distanceToLight +
         light.quadratic * distanceToLight * distanceToLight);

    vec3 ambient = light.ambient * albedo;
    vec3 diffuse = light.diffuse * diff * albedo;
    vec3 specular = light.specular * spec;

    return (ambient + diffuse + specular)
           * attenuation * light.intensity;
}


// Spot light
vec3 CalculateSpotLight(
    SpotLight light,
    vec3 normal,
    vec3 viewDir,
    vec3 albedo)
{
    vec3 lightOffset = light.position - fs_in.FragPos;
    float distanceToLight = length(lightOffset);
    vec3 lightDir = normalize(lightOffset);

    float diff = max(dot(normal, lightDir), 0.0);

    vec3 reflectDir = reflect(-lightDir, normal);
    float spec = pow(
        max(dot(viewDir, reflectDir), 0.0),
        32.0
    );

    float theta = dot(
        lightDir,
        normalize(-light.direction)
    );

    float innerCutoff = cos(radians(light.innerAngle));
    float outerCutoff = cos(radians(light.outerAngle));

    float coneIntensity = smoothstep(
        outerCutoff,
        innerCutoff,
        theta
    );

    float attenuation = 1.0 /
        (light.constant +
         light.linear * distanceToLight +
         light.quadratic * distanceToLight * distanceToLight);

    vec3 ambient = light.ambient * albedo;
    vec3 diffuse = light.diffuse * diff * albedo;
    vec3 specular = light.specular * spec;

    return (ambient + diffuse + specular)
           * attenuation * coneIntensity * light.intensity;
}


void main()
{
    vec3 normal = normalize(fs_in.Normal);
    vec3 viewDir = normalize(viewPos - fs_in.FragPos);

    // Sample the diffuse texture, or use the fallback material color.
    vec3 albedo = materialColor;

    if (useTexture)
    {
        albedo = texture(
            texture_diffuse1,
            fs_in.TexCoords
        ).rgb;
    }

    vec3 result = vec3(0.0);

    for (int i = 0; i < numDirectionalLights; ++i)
    {
        result += CalculateDirectionalLight(
            directionalLights[i], normal, viewDir, albedo
        );
    }

    for (int i = 0; i < numPointLights; ++i)
    {
        result += CalculatePointLight(
            pointLights[i], normal, viewDir, albedo
        );
    }

    for (int i = 0; i < numSpotLights; ++i)
    {
        result += CalculateSpotLight(
            spotLights[i], normal, viewDir, albedo
        );
    }

    FragColor = vec4(result, 1.0);
}
